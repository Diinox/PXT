cmake_minimum_required(VERSION 3.21)
project(tfs VERSION 1.0 LANGUAGES CXX)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
    message(FATAL_ERROR "Use an out-of-source build: cmake -S . -B build/linux")
endif()
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
option(USE_LUAJIT "Use LuaJIT (Lua 5.1 compatible)" ON)
option(ENABLE_PCH "Use precompiled headers" ON)
option(WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)

find_package(GMP REQUIRED)
find_package(PugiXML REQUIRED)
find_package(MySQL REQUIRED)
find_package(Threads REQUIRED)
find_package(Boost 1.53 CONFIG QUIET COMPONENTS system)
if(NOT Boost_FOUND)
    find_package(Boost 1.53 MODULE REQUIRED COMPONENTS system)
endif()
if(USE_LUAJIT)
    find_package(LuaJIT REQUIRED)
    set(TFS_LUA_TARGET LuaJIT::LuaJIT)
else()
    # Bypass the legacy local FindLua module and require the Lua 5.1 API.
    find_package(Lua 5.1 EXACT REQUIRED MODULE)
    if(NOT LUA_VERSION_STRING MATCHES "^5\\.1\\.")
        message(FATAL_ERROR "USE_LUAJIT=OFF requires Lua 5.1. Install Lua 5.1 or enable LuaJIT.")
    endif()
    add_library(TfsLua INTERFACE)
    target_include_directories(TfsLua INTERFACE ${LUA_INCLUDE_DIR})
    target_link_libraries(TfsLua INTERFACE ${LUA_LIBRARIES})
    set(TFS_LUA_TARGET TfsLua)
endif()
include(src/CMakeLists.txt)
add_executable(tfs ${tfs_SRC})
set_target_properties(tfs PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_link_libraries(tfs PRIVATE MySQL::Client ${TFS_LUA_TARGET}
    Boost::system PugiXML::PugiXML GMP::GMP Threads::Threads)
if(MSVC)
    target_compile_options(tfs PRIVATE /W3 /EHsc)
    target_compile_definitions(tfs PRIVATE _CRT_SECURE_NO_WARNINGS)
    if(WARNINGS_AS_ERRORS)
        target_compile_options(tfs PRIVATE /WX)
    endif()
elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(tfs PRIVATE -Wall -Wextra -pipe -fvisibility=hidden -fno-strict-aliasing)
    if(WARNINGS_AS_ERRORS)
        target_compile_options(tfs PRIVATE -Werror)
    endif()
endif()
if(WIN32)
    target_link_libraries(tfs PRIVATE ws2_32 mswsock)
endif()
if(ENABLE_PCH)
    target_precompile_headers(tfs PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/otpch.h")
endif()
